<script>
let data = JSON.parse(localStorage.getItem("modsData")) || {};
let currentGame = null;

function save() {
localStorage.setItem("modsData", JSON.stringify(data));
}

function addGame() {
let name = prompt("Game name");
if (!name) return;

if (!data[name]) data[name] = [];
currentGame = name;
render();
}

function setGame(name) {
currentGame = name;
render();
}

function addMod() {
if (!currentGame) return;

data[currentGame].push({
name: modName.value,
link: modLink.value,
category: modCategory.value,
status: modStatus.value,
notes: modNotes.value
});

modName.value = "";
modLink.value = "";
modNotes.value = "";

save();
render();
}

function render() {
let gameList = document.getElementById("gameList");
let modsDiv = document.getElementById("mods");
let search = document.getElementById("search").value.toLowerCase();

gameList.innerHTML = "";
modsDiv.innerHTML = "";

for (let game in data) {
let btn = document.createElement("button");
btn.innerText = game;
btn.onclick = () => setGame(game);
gameList.appendChild(btn);
}

if (!currentGame) return;
document.getElementById("title").innerText = currentGame;

data[currentGame]
.filter(m =>
m.name.toLowerCase().includes(search) ||
m.category.toLowerCase().includes(search) ||
currentGame.toLowerCase().includes(search)
)
.forEach(mod => {
let div = document.createElement("div");
div.className = "mod";

div.innerHTML = `
<b>${mod.name}</b><br>
<span class="small">${mod.category} | ${mod.status}</span><br>
<a href="${mod.link}" target="_blank">Link</a><br>
<span class="small">${mod.notes}</span>
`;

modsDiv.appendChild(div);
});

save();
}

render();
</script>
